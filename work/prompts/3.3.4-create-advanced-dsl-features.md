# Task 3.3.4: Create Advanced DSL Features

## Objective

Extend the basic `#[protocol]` macro to support advanced session type patterns including choices, branching, loops, recursion, and conditional protocols. This task implements the sophisticated DSL features that make complex protocol specification natural and intuitive.

## Background

Building on the basic attribute macro infrastructure from Task 3.3.3, this task adds support for the complex control flow patterns that make session types powerful. These features correspond to advanced session type constructs like `TChanChoice`, `TChanOffer`, `TChanRec`, and conditional branching.

This task represents the transition from simple linear protocols to the full expressiveness of session types while maintaining the natural, arrow-based syntax established in the basic implementation.

## Requirements

### 3. Conditional Protocols

Support conditional protocol execution:

```rust

/// protocol ConditionalAuth {
```

```rust
// In besedarium-macros/src/protocol_parser.rs

impl Parse for ProtocolFlow {
    fn parse(input: syn::parse::ParseStream) -> syn::Result<Self> {
        let lookahead = input.lookahead1();
        
        if lookahead.peek(Token![match]) {
            Ok(ProtocolFlow::Choice(input.parse()?))
        } else if lookahead.peek(Token![loop]) {
            Ok(ProtocolFlow::Loop(input.parse()?))
        } else if lookahead.peek(Token![if]) {
            Ok(ProtocolFlow::Conditional(input.parse()?))
        } else if lookahead.peek(Token![par]) {
            Ok(ProtocolFlow::Parallel(input.parse()?))
        } else if lookahead.peek(Token![end]) {
            input.parse::<Token![end]>()?;
            Ok(ProtocolFlow::End)
        } else {
            // Try to parse as message flow
            Ok(ProtocolFlow::MessageFlow(input.parse()?))
        }
    }
}

impl Parse for ChoiceFlow {
    fn parse(input: syn::parse::ParseStream) -> syn::Result<Self> {
        // Parse: Sender -> Receiver: ChoiceMessage { ... };
        // followed by: match { ... }
        
        let sender = input.parse::<Ident>()?;
        input.parse::<Token![->]>()?;
        let receiver = input.parse::<Ident>()?;
        input.parse::<Token![:]>()?;
        
        let message = input.parse::<ChoiceMessage>()?;
        input.parse::<Token![;]>()?;
        
        // Parse match block
        input.parse::<Token![match]>()?;
        let content;
        syn::braced!(content in input);
        
        let mut branches = Vec::new();
        while !content.is_empty() {
            branches.push(content.parse::<ChoiceBranch>()?);
            if content.peek(Token![,]) {
                content.parse::<Token![,]>()?;
            }
        }
        
        Ok(ChoiceFlow {
            sender,
            receiver,
            message,
            branches,
        })
    }
}

impl Parse for ChoiceMessage {
    fn parse(input: syn::parse::ParseStream) -> syn::Result<Self> {
        let name = input.parse::<Ident>()?;
        
        let content;
        syn::braced!(content in input);
        
        let mut variants = Vec::new();
        while !content.is_empty() {
            variants.push(content.parse::<ChoiceVariant>()?);
            if content.peek(Token![,]) {
                content.parse::<Token![,]>()?;
            }
        }
        
        Ok(ChoiceMessage { name, variants })
    }
}

impl Parse for ChoiceVariant {
    fn parse(input: syn::parse::ParseStream) -> syn::Result<Self> {
        let name = input.parse::<Ident>()?;
        
        let mut fields = Vec::new();
        if input.peek(syn::token::Paren) {
            let content;
            syn::parenthesized!(content in input);
            
            let field_list: Punctuated<MessageField, Token![,]> = 
                content.parse_terminated(MessageField::parse)?;
            fields = field_list.into_iter().collect();
        }
        
        Ok(ChoiceVariant { name, fields })
    }
}

impl Parse for ChoiceBranch {
    fn parse(input: syn::parse::ParseStream) -> syn::Result<Self> {
        // Parse: VariantName(field1, field2) => { ... }
        
        let variant = input.parse::<Ident>()?;
        
        let mut bound_fields = Vec::new();
        if input.peek(syn::token::Paren) {
            let content;
            syn::parenthesized!(content in input);
            
            let fields: Punctuated<Ident, Token![,]> = 
                content.parse_terminated(Ident::parse)?;
            bound_fields = fields.into_iter().collect();
        }
        
        input.parse::<Token![=>]>()?;
        
        let content;
        syn::braced!(content in input);
        
        let mut continuation = Vec::new();
        while !content.is_empty() {
            continuation.push(content.parse::<ProtocolFlow>()?);
        }
        
        Ok(ChoiceBranch {
            variant,
            bound_fields,
            continuation,
        })
    }
}

impl Parse for LoopFlow {
    fn parse(input: syn::parse::ParseStream) -> syn::Result<Self> {
        input.parse::<Token![loop]>()?;
        
        let content;
        syn::braced!(content in input);
        
        let mut body = Vec::new();
        while !content.is_empty() {
            body.push(content.parse::<ProtocolFlow>()?);
        }
        
        Ok(LoopFlow { body })
    }
}

impl Parse for ConditionalFlow {
    fn parse(input: syn::parse::ParseStream) -> syn::Result<Self> {
        input.parse::<Token![if]>()?;
        let condition = input.parse::<Ident>()?;
        
        let if_content;
        syn::braced!(if_content in input);
        
        let mut if_branch = Vec::new();
        while !if_content.is_empty() {
            if_branch.push(if_content.parse::<ProtocolFlow>()?);
        }
        
        let else_branch = if input.peek(Token![else]) {
            input.parse::<Token![else]>()?;
            
            let else_content;
            syn::braced!(else_content in input);
            
            let mut else_flows = Vec::new();
            while !else_content.is_empty() {
                else_flows.push(else_content.parse::<ProtocolFlow>()?);
            }
            
            Some(else_flows)
        } else {
            None
        };
        
        Ok(ConditionalFlow {
            condition,
            if_branch,
            else_branch,
        })
    }
}
```

```rust
fn generate_advanced_protocol_types(spec: &ProtocolSpec) -> syn::Result<proc_macro2::TokenStream> {
    let protocol_name = &spec.name;
    let roles = &spec.roles;
    
    // Analyze the protocol flows to determine session type structure
    let session_analysis = analyze_protocol_flows(&spec.flows)?;
    
    // Generate session types for each role
    let mut role_protocols = Vec::new();
    
    for role in roles {
        let role_protocol_name = quote::format_ident!("{}{}Protocol", protocol_name, role);
        let session_type = generate_session_type_for_role(role, &session_analysis)?;
        
        role_protocols.push(quote! {
            pub type #role_protocol_name = #session_type;
        });
    }
    
    Ok(quote! {
        #(#role_protocols)*
    })
}

fn analyze_protocol_flows(flows: &[ProtocolFlow]) -> syn::Result<SessionAnalysis> {
    let mut analysis = SessionAnalysis::new();
    
    for flow in flows {
        match flow {
            ProtocolFlow::MessageFlow(msg_flow) => {
                analysis.add_message_flow(msg_flow);
            }
            ProtocolFlow::Choice(choice_flow) => {
                analysis.add_choice_flow(choice_flow)?;
            }
            ProtocolFlow::Loop(loop_flow) => {
                analysis.add_loop_flow(loop_flow)?;
            }
            ProtocolFlow::Conditional(cond_flow) => {
                analysis.add_conditional_flow(cond_flow)?;
            }
            ProtocolFlow::Parallel(par_flow) => {
                analysis.add_parallel_flow(par_flow)?;
            }
            ProtocolFlow::End => {
                analysis.mark_end();
            }
        }
    }
    
    Ok(analysis)
}

fn generate_session_type_for_role(
    role: &Ident,
    analysis: &SessionAnalysis
) -> syn::Result<proc_macro2::TokenStream> {
    // Generate the session type based on the role's perspective
    // This involves complex analysis of the protocol flow
    
    let session_construction = construct_session_type(role, &analysis.flows)?;
    
    Ok(session_construction)
}

fn construct_session_type(
    role: &Ident,
    flows: &[AnalyzedFlow]
) -> syn::Result<proc_macro2::TokenStream> {
    match flows.len() {
        0 => Ok(quote! { ::besedarium::protocol::global::TChanEnd }),
        1 => construct_single_flow_type(role, &flows[0]),
        _ => construct_sequential_type(role, flows),
    }
}

fn construct_single_flow_type(
    role: &Ident,
    flow: &AnalyzedFlow
) -> syn::Result<proc_macro2::TokenStream> {
    match flow {
        AnalyzedFlow::Send { receiver, message, continuation } => {
            if role == &flow.sender() {
                let cont_type = construct_session_type(role, continuation)?;
                Ok(quote! {
                    ::besedarium::protocol::global::TChanSend<#message, #cont_type>
                })
            } else if role == receiver {
                let cont_type = construct_session_type(role, continuation)?;
                Ok(quote! {
                    ::besedarium::protocol::global::TChanRecv<#message, #cont_type>
                })
            } else {
                // Role not involved in this flow
                construct_session_type(role, continuation)
            }
        }
        AnalyzedFlow::Choice { sender, receiver, variants, branches } => {
            if role == sender {
                // Generate TChanChoice
                let choice_branches = generate_choice_branches(role, variants, branches)?;
                Ok(quote! {
                    ::besedarium::protocol::global::TChanChoice<#choice_branches>
                })
            } else if role == receiver {
                // Generate TChanOffer
                let offer_branches = generate_offer_branches(role, variants, branches)?;
                Ok(quote! {
                    ::besedarium::protocol::global::TChanOffer<#offer_branches>
                })
            } else {
                // Role not involved
                // Need to analyze branches to see if role is involved later
                construct_uninvolved_choice_type(role, branches)
            }
        }
        AnalyzedFlow::Loop { body, exit_condition } => {
            // Generate TChanRec for recursive protocols
            let loop_body = construct_session_type(role, body)?;
            Ok(quote! {
                ::besedarium::protocol::global::TChanRec<#loop_body>
            })
        }
        // ... other flow types
    }
}
```

```rust
fn generate_choice_types(choice_flow: &ChoiceFlow) -> syn::Result<proc_macro2::TokenStream> {
    let choice_name = &choice_flow.message.name;
    let variants = &choice_flow.message.variants;
    
    // Generate the choice enum
    let variant_defs = variants.iter().map(|variant| {
        let var_name = &variant.name;
        if variant.fields.is_empty() {
            quote! { #var_name }
        } else {
            let field_types = variant.fields.iter().map(|f| &f.field_type);
            quote! { #var_name(#(#field_types),*) }
        }
    });
    
    let choice_enum = quote! {
        #[derive(Debug, Clone, PartialEq)]
        pub enum #choice_name {
            #(#variant_defs,)*
        }
        
        impl ::besedarium::protocol::foundation::Message for #choice_name {
            type Metadata = ::besedarium::protocol::foundation::StandardMetadata;
        }
    };
    
    // Generate individual message types for each variant
    let variant_types = variants.iter().map(|variant| {
        let var_name = &variant.name;
        if variant.fields.is_empty() {
            quote! {
                #[derive(Debug, Clone, Copy, PartialEq, Eq)]
                pub struct #var_name;
                
                impl ::besedarium::protocol::foundation::Message for #var_name {
                    type Metadata = ::besedarium::protocol::foundation::StandardMetadata;
                }
            }
        } else {
            let field_defs = variant.fields.iter().map(|f| {
                let name = &f.name;
                let ty = &f.field_type;
                quote! { pub #name: #ty }
            });
            
            quote! {
                #[derive(Debug, Clone, PartialEq)]
                pub struct #var_name {
                    #(#field_defs,)*
                }
                
                impl ::besedarium::protocol::foundation::Message for #var_name {
                    type Metadata = ::besedarium::protocol::foundation::StandardMetadata;
                }
            }
        }
    });
    
    Ok(quote! {
        #choice_enum
        #(#variant_types)*
    })
}
```
