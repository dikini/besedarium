## Context and Requirements

### Runtime System Design Principles

The runtime system must bridge the gap between compile-time protocol verification and runtime execution:

1. **Type-Safe Execution**: Runtime state must correspond to compile-time protocol types
2. **Protocol Adherence**: Runtime must enforce protocol ordering and prevent violations
3. **Error Recovery**: Clear error types for protocol violations and communication failures
4. **Performance**: Minimal runtime overhead while maintaining safety guarantees
5. **Async Support**: Integration with async/await for non-blocking communication

### Key Components to Implement

1. **State Machine**: Track protocol execution state and validate transitions
2. **Channel Communication**: Typed message passing that respects protocol constraints
3. **Error Types**: Comprehensive error hierarchy for runtime failures
4. **Session Management**: Lifecycle management for protocol sessions

```rust
/// Represents the current execution state of a protocol
#[derive(Debug, Clone)]
pub struct ProtocolState<P> {
    protocol: PhantomData<P>,
    metadata: CommMetadata,
    step_count: usize,
    is_complete: bool,
}

impl<P> ProtocolState<P> {
    /// Create initial state for protocol P
    pub fn new(metadata: CommMetadata) -> Self {
        Self {
            protocol: PhantomData,
            metadata,
            step_count: 0,
            is_complete: false,
        }
    }
    
    /// Get current metadata
    pub fn metadata(&self) -> &CommMetadata {
        &self.metadata
    }
    
    /// Check if protocol execution is complete
    pub fn is_complete(&self) -> bool {
        self.is_complete
    }
    
    /// Get execution step count
    pub fn step_count(&self) -> usize {
        self.step_count
    }
}

/// Represents a valid state transition in protocol execution
pub trait StateTransition<From, To> {
    type Error;
    
    /// Attempt to transition from one protocol state to another
    fn transition(
        from: ProtocolState<From>
    ) -> Result<ProtocolState<To>, Self::Error>;
}

/// Execution context for protocol operations
#[derive(Debug)]
pub struct ExecutionContext {
    session_id: String,
    role: Box<dyn Role>,
    start_time: std::time::Instant,
    metadata: std::collections::HashMap<String, String>,
}

impl ExecutionContext {
    pub fn new(session_id: String, role: Box<dyn Role>) -> Self {
        Self {
            session_id,
            role,
            start_time: std::time::Instant::now(),
            metadata: std::collections::HashMap::new(),
        }
    }
    
    pub fn session_id(&self) -> &str {
        &self.session_id
    }
    
    pub fn role(&self) -> &dyn Role {
        self.role.as_ref()
    }
    
    pub fn elapsed(&self) -> std::time::Duration {
        self.start_time.elapsed()
    }
    
    pub fn set_metadata(&mut self, key: String, value: String) {
        self.metadata.insert(key, value);
    }
    
    pub fn get_metadata(&self, key: &str) -> Option<&String> {
        self.metadata.get(key)
    }
```

### Step 3: Implement Channel Communication

#### Typed Channel Infrastructure

```rust
use std::marker::PhantomData;
use std::pin::Pin;
use std::task::{Context, Poll};
use futures::{Future, Sink, Stream};
use tokio::sync::mpsc;
use serde::{Serialize, Deserialize};

use crate::protocol::foundation::{Message, Role, CommMetadata};
use crate::runtime::error::{RuntimeError, CommunicationError};

/// A typed channel that enforces protocol message types
#[derive(Debug)]
pub struct TypedChannel<Msg> {
    sender: mpsc::UnboundedSender<ChannelMessage<Msg>>,
    receiver: mpsc::UnboundedReceiver<ChannelMessage<Msg>>,
    metadata: CommMetadata,
    _msg: PhantomData<Msg>,
}

/// Wrapper for messages with metadata
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ChannelMessage<Msg> {
    pub message: Msg,
    pub metadata: CommMetadata,
    pub timestamp: std::time::SystemTime,
    pub sequence: u64,
}

impl<Msg> TypedChannel<Msg>
where
    Msg: Message + Send + 'static,
{
    /// Create a new typed channel pair
    pub fn new(metadata: CommMetadata) -> (Self, Self) {
        let (tx1, rx1) = mpsc::unbounded_channel();
        let (tx2, rx2) = mpsc::unbounded_channel();
        
        let channel1 = TypedChannel {
            sender: tx1,
            receiver: rx2,
            metadata: metadata.clone(),
            _msg: PhantomData,
        };
        
        let channel2 = TypedChannel {
            sender: tx2,
            receiver: rx1,
            metadata,
            _msg: PhantomData,
        };
        
        (channel1, channel2)
    }
    
    /// Send a message through the channel
    pub async fn send(&mut self, message: Msg) -> Result<(), RuntimeError> {
        let channel_msg = ChannelMessage {
            message,
            metadata: self.metadata.clone(),
            timestamp: std::time::SystemTime::now(),
            sequence: 0, // TODO: Implement proper sequencing
        };
        
        self.sender
            .send(channel_msg)
            .map_err(|_| RuntimeError::Communication(
                CommunicationError::ChannelClosed
            ))
    }
    
    /// Receive a message from the channel
    pub async fn recv(&mut self) -> Result<Msg, RuntimeError> {
        match self.receiver.recv().await {
            Some(channel_msg) => Ok(channel_msg.message),
            None => Err(RuntimeError::Communication(
                CommunicationError::ChannelClosed
            )),
        }
    }
    
    /// Check if channel is closed
    pub fn is_closed(&self) -> bool {
        self.sender.is_closed()
    }
    
    /// Get channel metadata
    pub fn metadata(&self) -> &CommMetadata {
        &self.metadata
    }
}

/// Convenient type alias for channel pairs
pub type ChannelPair<Msg> = (TypedChannel<Msg>, TypedChannel<Msg>);

/// Trait for types that can send messages
pub trait MessageSend<Msg> {
    type Error;
    type SendFuture: Future<Output = Result<(), Self::Error>>;
    
    fn send(&mut self, message: Msg) -> Self::SendFuture;
}

/// Trait for types that can receive messages
pub trait MessageRecv<Msg> {
    type Error;
    type RecvFuture: Future<Output = Result<Msg, Self::Error>>;
    
    fn recv(&mut self) -> Self::RecvFuture;
}

impl<Msg> MessageSend<Msg> for TypedChannel<Msg>
where
    Msg: Message + Send + 'static,
{
    type Error = RuntimeError;
    type SendFuture = Pin<Box<dyn Future<Output = Result<(), Self::Error>> + Send>>;
    
    fn send(&mut self, message: Msg) -> Self::SendFuture {
        Box::pin(self.send(message))
    }
}

impl<Msg> MessageRecv<Msg> for TypedChannel<Msg>
where
    Msg: Message + Send + 'static,
{
    type Error = RuntimeError;
    type RecvFuture = Pin<Box<dyn Future<Output = Result<Msg, Self::Error>> + Send>>;
    
    fn recv(&mut self) -> Self::RecvFuture {
        Box::pin(self.recv())
    }
}
```

#### Session Lifecycle Management

```rust
use std::collections::HashMap;
use std::sync::{Arc, Mutex};
use std::time::{Duration, Instant};
use uuid::Uuid;

use crate::protocol::{
    foundation::{Role, CommMetadata},
    global::GlobalProtocol,
    local::LocalProtocol
};
use crate::runtime::{
    error::{RuntimeError, SessionError},
    state::{ProtocolState, ExecutionContext},
    channel::TypedChannel
};

/// Configuration for session creation
#[derive(Debug, Clone)]
pub struct SessionConfig {
    pub timeout: Duration,
    pub max_message_size: usize,
    pub enable_logging: bool,
    pub retry_attempts: usize,
    pub metadata: HashMap<String, String>,
}

impl Default for SessionConfig {
    fn default() -> Self {
        Self {
            timeout: Duration::from_secs(30),
            max_message_size: 1024 * 1024, // 1MB
            enable_logging: true,
            retry_attempts: 3,
            metadata: HashMap::new(),
        }
    }
}

/// A runtime session for protocol execution
#[derive(Debug)]
pub struct Session<P> {
    id: String,
    state: ProtocolState<P>,
    context: ExecutionContext,
    config: SessionConfig,
    created_at: Instant,
    last_activity: Instant,
}

impl<P> Session<P>
where
    P: GlobalProtocol + Send + Sync + 'static,
{
    /// Create a new session
    pub fn new(
        role: Box<dyn Role>,
        metadata: CommMetadata,
        config: SessionConfig,
    ) -> Self {
        let id = Uuid::new_v4().to_string();
        let state = ProtocolState::new(metadata);
        let context = ExecutionContext::new(id.clone(), role);
        let now = Instant::now();
        
        Self {
            id,
            state,
            context,
            config,
            created_at: now,
            last_activity: now,
        }
    }
    
    /// Get session ID
    pub fn id(&self) -> &str {
        &self.id
    }
    
    /// Get current protocol state
    pub fn state(&self) -> &ProtocolState<P> {
        &self.state
    }
    
    /// Get execution context
    pub fn context(&self) -> &ExecutionContext {
        &self.context
    }
    
    /// Check if session has expired
    pub fn is_expired(&self) -> bool {
        self.last_activity.elapsed() > self.config.timeout
    }
    
    /// Update last activity timestamp
    pub fn update_activity(&mut self) {
        self.last_activity = Instant::now();
    }
    
    /// Get session duration
    pub fn duration(&self) -> Duration {
        self.created_at.elapsed()
    }
    
    /// Check if session is complete
    pub fn is_complete(&self) -> bool {
        self.state.is_complete()
    }
}

/// Handle for managing session lifecycle
#[derive(Debug)]
pub struct SessionHandle<P> {
    session: Arc<Mutex<Session<P>>>,
}

impl<P> SessionHandle<P>
where
    P: GlobalProtocol + Send + Sync + 'static,
{
    /// Create a new session handle
    pub fn new(session: Session<P>) -> Self {
        Self {
            session: Arc::new(Mutex::new(session)),
        }
    }
    
    /// Execute a closure with mutable access to the session
    pub fn with_session<F, R>(&self, f: F) -> Result<R, RuntimeError>
    where
        F: FnOnce(&mut Session<P>) -> Result<R, RuntimeError>,
    {
        let mut session = self.session
            .lock()
            .map_err(|_| RuntimeError::Session(
                SessionError::InvalidState {
                    state: "poisoned".to_string()
                }
            ))?;
            
        if session.is_expired() {
            return Err(RuntimeError::Session(
                SessionError::Expired {
                    session_id: session.id().to_string()
                }
            ));
        }
        
        session.update_activity();
        f(&mut *session)
    }
    
    /// Execute a closure with read-only access to the session
    pub fn with_session_read<F, R>(&self, f: F) -> Result<R, RuntimeError>
    where
        F: FnOnce(&Session<P>) -> Result<R, RuntimeError>,
    {
        let session = self.session
            .lock()
            .map_err(|_| RuntimeError::Session(
                SessionError::InvalidState {
                    state: "poisoned".to_string()
                }
            ))?;
            
        if session.is_expired() {
            return Err(RuntimeError::Session(
                SessionError::Expired {
                    session_id: session.id().to_string()
                }
            ));
        }
        
        f(&*session)
    }
    
    /// Clone the session handle
    pub fn clone_handle(&self) -> Self {
        Self {
            session: Arc::clone(&self.session),
        }
    }
}

/// Session manager for handling multiple concurrent sessions
#[derive(Debug)]
pub struct SessionManager {
    sessions: Arc<Mutex<HashMap<String, Box<dyn std::any::Any + Send + Sync>>>>,
    config: SessionConfig,
}

impl SessionManager {
    /// Create a new session manager
    pub fn new(config: SessionConfig) -> Self {
        Self {
            sessions: Arc::new(Mutex::new(HashMap::new())),
            config,
        }
    }
    
    /// Create a new session
    pub fn create_session<P>(
        &self,
        role: Box<dyn Role>,
        metadata: CommMetadata,
    ) -> Result<SessionHandle<P>, RuntimeError>
    where
        P: GlobalProtocol + Send + Sync + 'static,
    {
        let session = Session::new(role, metadata, self.config.clone());
        let handle = SessionHandle::new(session);
        
        let mut sessions = self.sessions
            .lock()
            .map_err(|_| RuntimeError::Session(
                SessionError::InvalidState {
                    state: "poisoned".to_string()
                }
            ))?;
            
        let session_id = handle.with_session_read(|s| Ok(s.id().to_string()))?;
        
        if sessions.contains_key(&session_id) {
            return Err(RuntimeError::Session(
                SessionError::AlreadyExists { session_id }
            ));
        }
        
        sessions.insert(session_id, Box::new(handle.clone_handle()));
        Ok(handle)
    }
    
    /// Remove expired sessions
    pub fn cleanup_expired(&self) -> Result<usize, RuntimeError> {
        let mut sessions = self.sessions
            .lock()
            .map_err(|_| RuntimeError::Session(

            true
        });
        
        Ok(initial_count - sessions.len())
    }
```
