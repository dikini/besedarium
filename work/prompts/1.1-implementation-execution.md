//! Communication metadata system with channel and label support

use std::marker::PhantomData;

/// Channel identifier trait
pub trait ChanId: 'static + Clone + std::fmt::Debug + Send + Sync {}

/// Message label trait
pub trait MsgLbl: 'static + Clone + std::fmt::Debug + Send + Sync {}

/// Communication metadata combining channel and label information
#[derive(Clone, Debug)]
pub struct CommMetadata<C: ChanId, L: MsgLbl> {
    _phantom: PhantomData<(C, L)>,
}

impl<C: ChanId, L: MsgLbl> CommMetadata<C, L> {
    pub fn new() -> Self {
        Self {
            _phantom: PhantomData,
        }
    }
}

// Standard channel types
#[derive(Clone, Debug, PartialEq, Eq)]

impl ChanId for DefaultChan {}

//! Global protocol type for channel sending

use crate::protocol::foundation::{GlobalProtocol, Role};
use crate::protocol::metadata::{CommMetadata, ChanId, MsgLbl};
use crate::protocol::action_io::{ActionIOTMarker, SupportsActionIO};
use std::marker::PhantomData;

/// Global protocol type for sending a message from sender to receiver
#[derive(Clone, Debug)]
pub struct TChanSend<M, S, R, T, IO, P>
where
    M: CommMetadata,
    S: Role,
    R: Role,
    IO: ActionIOTMarker,
    P: GlobalProtocol,
{
    _phantom: PhantomData<(M, S, R, T, IO, P)>,
}

impl<M, S, R, T, IO, P> GlobalProtocol for TChanSend<M, S, R, T, IO, P>
where
    M: CommMetadata,
    S: Role,
    R: Role,
    IO: ActionIOTMarker,
    P: GlobalProtocol,
{}

impl<M, S, R, T, IO, P> SupportsActionIO<IO> for TChanSend<M, S, R, T, IO, P>
where
    M: CommMetadata,
    S: Role,
    R: Role,
    IO: ActionIOTMarker,
    P: GlobalProtocol + SupportsActionIO<IO>,
{}

/// Type-level boolean trait
pub trait TypeBool: 'static {}

pub struct TTrue;

/// Type-level false

    type Output: TypeBool;
}

    type Output: TypeBool;
}

    type Output = TFalse;
}
